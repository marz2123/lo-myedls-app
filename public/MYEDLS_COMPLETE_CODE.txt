// === src/contexts/AppModeContext.tsx ===
import { createContext, useContext, useState, useEffect, ReactNode } from "react";

export type AppMode = 'selector' | 'classic' | 'aladin';

interface AppModeContextType {
  appMode: AppMode;
  setAppMode: (mode: AppMode) => void;
}

const AppModeContext = createContext<AppModeContextType | undefined>(undefined);

export const AppModeProvider = ({ children }: { children: ReactNode }) => {
  const [appMode, setAppModeState] = useState<AppMode>('selector');

  useEffect(() => {
    const savedMode = localStorage.getItem('myedls-app-mode') as AppMode | null;
    if (savedMode === 'classic' || savedMode === 'aladin') {
      setAppModeState(savedMode);
    }
  }, []);

  const setAppMode = (mode: AppMode) => {
    if (mode === 'selector') {
      localStorage.removeItem('myedls-app-mode');
    } else {
      localStorage.setItem('myedls-app-mode', mode);
    }
    setAppModeState(mode);
  };

  return (
    <AppModeContext.Provider value={{ appMode, setAppMode }}>
      {children}
    </AppModeContext.Provider>
  );
};

export const useAppMode = () => {
  const context = useContext(AppModeContext);
  if (!context) throw new Error('useAppMode must be used within AppModeProvider');
  return context;
};

// === src/hooks/usePropertyStructure.ts ===
import { useState, useEffect, useCallback } from 'react';
import { supabase } from '@/integrations/supabase/client';
import { toast } from 'sonner';

export const usePropertyStructure = (projectId: string | undefined) => {
  const [parts, setParts] = useState([]);
  const [locations, setLocations] = useState([]);
  const [loading, setLoading] = useState(true);

  const loadStructure = useCallback(async () => {
    if (!projectId) return;
    setLoading(true);
    try {
      const { data: partsData } = await supabase
        .from('property_parts')
        .select('*')
        .eq('project_id', projectId)
        .order('order_index');
      setParts(partsData || []);

      const { data: locsData } = await supabase
        .from('property_locations')
        .select('*')
        .eq('project_id', projectId)
        .order('order_index');
      setLocations(locsData || []);
    } finally {
      setLoading(false);
    }
  }, [projectId]);

  useEffect(() => { loadStructure(); }, [loadStructure]);

  const createPart = async (partType, name) => {
    const { data, error } = await supabase
      .from('property_parts')
      .insert({ project_id: projectId, part_type: partType, name, order_index: parts.length })
      .select().single();
    if (!error) setParts(prev => [...prev, data]);
    return data;
  };

  const createLocation = async (partId, name) => {
    const { data, error } = await supabase
      .from('property_locations')
      .insert({ project_id: projectId, part_id: partId, name, order_index: 0 })
      .select().single();
    if (!error) setLocations(prev => [...prev, data]);
    return data;
  };

  return { parts, locations, loading, loadStructure, createPart, createLocation };
};

// === src/hooks/useVisitSequences.ts ===
import { useState, useEffect, useCallback } from 'react';
import { supabase } from '@/integrations/supabase/client';
import { toast } from 'sonner';

export const useVisitSequences = (projectId: string | undefined) => {
  const [sequences, setSequences] = useState([]);
  const [problems, setProblems] = useState([]);
  const [tasks, setTasks] = useState([]);
  const [loading, setLoading] = useState(true);

  const loadSequences = useCallback(async () => {
    if (!projectId) return;
    setLoading(true);
    try {
      const { data: seqData } = await supabase
        .from('visit_sequences')
        .select('*')
        .eq('project_id', projectId)
        .order('started_at', { ascending: false });
      setSequences(seqData || []);

      const { data: probData } = await supabase
        .from('identified_problems')
        .select('*')
        .eq('project_id', projectId);
      setProblems(probData || []);

      const { data: taskData } = await supabase
        .from('problem_tasks')
        .select('*, task_families(name), task_categories(name), task_subcategories(name)')
        .eq('project_id', projectId);
      setTasks(taskData || []);
    } finally {
      setLoading(false);
    }
  }, [projectId]);

  useEffect(() => { loadSequences(); }, [loadSequences]);

  const startSequence = async (partId, locationId) => {
    const { data: { user } } = await supabase.auth.getUser();
    if (!user) return null;
    
    const { data, error } = await supabase
      .from('visit_sequences')
      .insert({
        project_id: projectId,
        part_id: partId,
        location_id: locationId,
        user_id: user.id,
        status: 'recording',
        photos: [],
        detected_zones: [],
      })
      .select().single();
    if (!error) setSequences(prev => [data, ...prev]);
    return data;
  };

  const analyzeSequence = async (sequenceId, options = {}) => {
    const sequence = sequences.find(s => s.id === sequenceId);
    if (!sequence) return null;

    const { data, error } = await supabase.functions.invoke('analyze-visit-sequence', {
      body: {
        sequenceId,
        projectId,
        partId: sequence.part_id,
        locationId: sequence.location_id,
        ...options,
      }
    });
    if (!error) await loadSequences();
    return data;
  };

  return { sequences, problems, tasks, loading, loadSequences, startSequence, analyzeSequence };
};

    doc.text(`Date: ${new Date().toLocaleDateString()}`, 20, 30);

    let y = 50;
    tasks.forEach((task, i) => {
      if (y > 270) { doc.addPage(); y = 20; }

      y += 8;
    });
    
    doc.save(`EDL-Report-${new Date().toISOString().split('T')[0]}.pdf`);
  };

  return (
    <Button onClick={generatePDF} variant="outline">
      <FileDown className="w-4 h-4 mr-2" />
      Exporter PDF
    </Button>
  );
};

// === supabase/functions/analyze-visit-sequence/index.ts ===
import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.84.0";

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders });

  try {
    const supabase = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
    );

    const { sequenceId, projectId, transcription, photoUrls } = await req.json();

    // Update status to processing
    await supabase.from('visit_sequences').update({ status: 'processing' }).eq('id', sequenceId);

    // Call AI for analysis
    const lovableApiKey = Deno.env.get('LOVABLE_API_KEY');
    let analysisResult = { detected_zones: ['murs', 'sol'], condition: 'bon', problems: [] };

    if (lovableApiKey) {
      const response = await fetch('https://ai.gateway.lovable.dev/v1/chat/completions', {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${lovableApiKey}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: 'google/gemini-2.5-flash',
          messages: [
            { role: 'system', content: 'Expert diagnostic immobilier. JSON uniquement.' },
            { role: 'user', content: `Analyse: ${transcription}` }
          ],
        }),
      });
      const data = await response.json();
      const content = data.choices?.[0]?.message?.content || '';
      const jsonMatch = content.match(/\{[\s\S]*\}/);
      if (jsonMatch) analysisResult = JSON.parse(jsonMatch[0]);
    }

    // Update sequence with results
    await supabase.from('visit_sequences').update({
      status: 'completed',
      ai_analysis: analysisResult,
      detected_condition: analysisResult.condition,
      detected_zones: analysisResult.detected_zones
    }).eq('id', sequenceId);

    return new Response(JSON.stringify({ success: true, analysis: analysisResult }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), { status: 500, headers: corsHeaders });
  }
});
